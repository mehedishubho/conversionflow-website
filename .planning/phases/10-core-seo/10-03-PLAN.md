```typescript
let sitemapOverrides: Record<string, string> = {};
try {
  const rows = await db.select().from(settings).where(inArray(settings.key, [
    "seo_sitemap_enabled", "seo_sitemap_pages", "seo_sitemap_blog",
    "seo_sitemap_docs", "seo_sitemap_landing", "seo_sitemap_excludes",
    "seo_sitemap_auto_regenerate"
  ]));
  for (const row of rows) sitemapOverrides[row.key] = row.value;
} catch { /* fallback to defaults */ }
```

When "false" or not set, use Next.js default caching (`revalidate = 3600` or omit for default behavior). This satisfies SITM-03 -- auto-regeneration means the sitemap always reflects current content.
