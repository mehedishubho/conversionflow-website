      function CopyButton({ text, label }: { text: string; label: string }) {
        const [copied, setCopied] = useState(false);
        const handleCopy = async () => {
          try {
            await navigator.clipboard.writeText(text);
          } catch {
            // Fallback for older browsers
            const textarea = document.createElement("textarea");
            textarea.value = text;
            document.body.appendChild(textarea);
            textarea.select();
            document.execCommand("copy");
            document.body.removeChild(textarea);
          }
          setCopied(true);
          setTimeout(() => setCopied(false), 2000);
        };
        return (
          <button
            onClick={handleCopy}
            className="p-1.5 rounded-md hover:bg-gray-100 dark:hover:bg-gray-800 transition-colors"
            aria-label={label}
          >
            {copied ? (

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Success page -> getOrderDetails | Authenticated user session. Returns only the user's own orders. |
| Email -> Resend API | API token shown in email. TLS in transit. Email delivered to verified address. |
| Clipboard API | Browser-native clipboard access. No security concern — user-initiated action. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-17-08 | Information Disclosure | API token in email | mitigate | API token shown once in email only. Not stored in DB (only hash). Success page does NOT display the plaintext token. |
| T-17-09 | Information Disclosure | License key on success page | accept | License key is not a secret — it's a product activation credential the customer owns. Displayed only to the authenticated owner. |
| T-17-10 | Tampering | getOrderDetails | mitigate | Session auth check + user-scoped query (`eq(orders.userId, session.user.id)`) — user can only see their own orders. |
