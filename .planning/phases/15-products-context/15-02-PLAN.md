import { headers } from "next/headers";
import { redirect } from "next/navigation";

import { createAuditLog } from "@/lib/audit";
import { nanoid } from "nanoid";

async function requireAdmin() {
  const session = await auth.api.getSession({ headers: await headers() });
  if (!session) redirect("/login");
  const role = (session.user as Record<string, unknown>).role as string;
  if (role !== "admin" && role !== "super_admin") redirect("/admin/dashboard");
  return { session, userId: session.user.id, role };
}

**Product Actions:**

1. `createProduct(formData: FormData)` — Extract `name` (required) and `description` (optional) from formData. Generate slug from name using `name.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '')`. Validate name is non-empty. Insert into `products` table. Create audit log with action `"product.created"`, targetType `"product"`. Return `{ success: true, productId: product.id }` or `{ error: string }`.

2. `updateProduct(productId: string, formData: FormData)` — Extract `name`, `description`, `currentVersion` from formData. Regenerate slug if name changed. Update in `products` table. Audit log `"product.updated"`. Return `{ success: true }` or `{ error: string }`.

3. `deleteProduct(productId: string)` — Delete from products table (cascade will delete versions and plans). Audit log `"product.deleted"`. Return `{ success: true }` or `{ error: string }`.

**Version Actions:**

4. `createVersion(productId: string, formData: FormData)` — Extract `version` (required semver string), `downloadUrl` (optional), `changelog` (optional) from formData. Insert into `productVersions` with status "draft". Audit log `"product.version.created"` with targetType `"product_version"`. Return `{ success: true, versionId }`.

5. `updateVersion(versionId: string, formData: FormData)` — Extract `version`, `downloadUrl`, `changelog`, `status` from formData. If status changes to "stable", also set `releasedAt` to `new Date()` (matching domain entity logic). Update in productVersions. Audit log `"product.version.updated"`. Return `{ success: true }`.

6. `releaseVersion(versionId: string)` — Set status to "stable" and releasedAt to new Date(). Audit log `"product.version.released"`. Return `{ success: true }`.

**Plan Actions:**

7. `createPlan(productId: string, formData: FormData)` — Extract all plan fields: `name` (required), `slug` (auto-generate from name), `description` (optional), `priceBDT` (required, parse as int), `priceUSD` (required, parse as int), `licenseType` (required, "lifetime" or "subscription"), `billingCycle` (optional, "monthly"/"yearly"/"custom"), `billingDurationMonths` (optional, parse as int), `maxActivations` (required, parse as int, 0 = unlimited), `features` (JSON string from form, parse as Record<string, boolean>), `sortOrder` (optional, parse as int).
   - Validate: if licenseType === "lifetime", billingCycle and billingDurationMonths must be null/undefined
   - Validate: if licenseType === "subscription", billingCycle must be provided
   - Validate: features must be a flat object with boolean values only
   - Insert into productPlans. Audit log `"product.plan.created"` with targetType `"product_plan"`. Return `{ success: true, planId }`.

8. `updatePlan(planId: string, formData: FormData)` — Same field extraction and validation as createPlan. Update in productPlans. Audit log `"product.plan.updated"`. Return `{ success: true }`.

9. `deletePlan(planId: string)` — Delete from productPlans. Audit log `"product.plan.deleted"`. Return `{ success: true }`.

All actions must:
- Call `requireAdmin()` first
- Use try/catch wrapping the database operation
- Return `{ error: "message" }` on failure
- Use `createAuditLog()` for all mutations
- Use explicit field extraction from FormData (never spread formData into DB insert)
