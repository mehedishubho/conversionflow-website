```typescript
static fromDatabase(value: string): LicenseKey {
  if (typeof value !== "string" || value.trim().length === 0) {
    throw new Error("License key cannot be empty");
  }
  const cleaned = value.toUpperCase().replace(/[^A-Z0-9]/g, "");
  const ambiguous = /[01OIL]/;
  if (ambiguous.test(cleaned)) {
    throw new Error("License key contains invalid characters");
  }
  return new LicenseKey(cleaned);
}
```

**Part F: Create LicenseKeyGenerator** at `src/modules/licensing/domain/services/LicenseKeyGenerator.ts`

This is the CRITICAL security-sensitive service. Use rejection sampling per D-03 (31-char charset):

```typescript
import crypto from "crypto";
import { LicenseKey } from "@/shared/domain/valueObjects/LicenseKey";

const CHARSET = "ABCDEFGHJKMNPQRSTUVWXYZ23456789";
const CHARSET_LENGTH = 31;
const KEY_LENGTH = 20;
const REJECTION_THRESHOLD = Math.floor(256 / CHARSET_LENGTH) * CHARSET_LENGTH; // 248

export class LicenseKeyGenerator {
  static generate(): LicenseKey {
    const body = Buffer.alloc(KEY_LENGTH);
    let written = 0;
    while (written < KEY_LENGTH) {
      const bytes = crypto.randomBytes(KEY_LENGTH * 2);
      for (let i = 0; i < bytes.length && written < KEY_LENGTH; i++) {
        const byte = bytes[i];
        if (byte < REJECTION_THRESHOLD) {
          body[written] = CHARSET.charCodeAt(byte % CHARSET_LENGTH);
          written++;
        }
      }
    }
    return LicenseKey.create(body.toString("ascii"));
  }

  static generateFormatted(): string {
    return `CF-${this.generate().formatted}`;
  }
}
```

**Part G: Create ApiTokenGenerator** at `src/modules/licensing/domain/services/ApiTokenGenerator.ts`

```typescript
import crypto from "crypto";
import { nanoid } from "nanoid";

export class ApiTokenGenerator {
  static generate(): { plaintext: string; hash: string } {
    const token = `cf_live_${nanoid(32)}`;
    const hash = crypto.createHash("sha256").update(token).digest("hex");
    return { plaintext: token, hash };
  }

  static validate(plaintext: string, storedHash: string): boolean {
    if (!plaintext || !storedHash) return false;
    const plaintextHash = crypto.createHash("sha256").update(plaintext).digest("hex");
    const a = Buffer.from(plaintextHash, "hex");
    const b = Buffer.from(storedHash, "hex");
    if (a.length !== b.length) return false;
    return crypto.timingSafeEqual(a, b);
  }
}
```

**Part H: Create VerificationTokenIssuer** at `src/modules/licensing/domain/services/VerificationTokenIssuer.ts`

```typescript
import crypto from "crypto";
import { redis, kvGet, kvSet, kvDelete } from "@/lib/redis";

const VERIFICATION_TOKEN_PREFIX = "verify:";
const VERIFICATION_TTL_SECONDS = 86400;

export class VerificationTokenIssuer {
  static async issue(licenseId: string, domain: string): Promise<string> {
    const token = crypto.randomBytes(16).toString("hex");
    const key = `${VERIFICATION_TOKEN_PREFIX}${token}`;
    await kvSet(key, JSON.stringify({ licenseId, domain }), VERIFICATION_TTL_SECONDS);
    return token;
  }

  static async consume(
    token: string,
    expectedLicenseId: string,
    expectedDomain: string,
  ): Promise<boolean> {
    const key = `${VERIFICATION_TOKEN_PREFIX}${token}`;
    if (redis) {
      const value = await redis.getdel(key);
      if (!value) return false;
      const parsed = JSON.parse(value);
      return parsed.licenseId === expectedLicenseId && parsed.domain === expectedDomain;
    }
    // Memory fallback
    const value = await kvGet(key);
    if (!value) return false;
    await kvDelete(key);
    const parsed = JSON.parse(value);
    return parsed.licenseId === expectedLicenseId && parsed.domain === expectedDomain;
  }
}
```

|----------|-------------|
| crypto.randomBytes → LicenseKey | Cryptographic random source feeding key generation; compromise = predictable keys |
| API token plaintext → SHA-256 hash | Hash boundary; timing attacks on comparison leak token validity |
